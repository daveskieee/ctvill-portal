The CTVILL Construction Project Management System (React, TypeScript, Vite, Tailwind CSS,

Key Vulnerabilities Resolved:
1. INSECURE CLIENT HEADERS ELIMINATED: Previously, endpoints trusted 'x-user-id' and
   'x-user-role' headers supplied by the client. All client-supplied privilege headers
   are now completely ignored.
2. AUTOMATIC ADMIN FALLBACKS REMOVED: Unauthenticated or unmapped roles previously

   HTTP 401 Unauthorized, and unmapped roles return HTTP 403 Forbidden (mapped to UNAUTHORIZED tier 0).

   stored as SHA-256 hashes in a dedicated 'sessions' table. Sessions are transmitted via
   an HttpOnly, SameSite=Lax cookie ('ctvill_session').
4. STRICT SEPARATION OF DUTIES (SoD): Timekeepers cannot finalize or approve payroll
   (enforced with 403 FORBIDDEN_SOD). Finance users cannot create Administrator accounts.
5. INSECURE DIRECT OBJECT REFERENCE (IDOR) PREVENTION: Project queries and mutations
   are strictly scoped against req.user.projectIds. Users cannot access projects outside
   their authorized assignments.
6. HARDCODED DEMO BYPASSES REMOVED: Demo password strings and login bypasses were removed.
   All logins strictly verify against scrypt password hashes. Password updates require minimum
   8 characters (letters + numbers) and automatically revoke all other existing sessions.
7. CROSS-SITE REQUEST FORGERY (CSRF) DEFENSE: State-changing requests (POST, PUT, PATCH, DELETE)
   require the 'X-CSRF-Token' header matching the 'ctvill_csrf' cookie.
8. OWASP HEADERS & DOTFILE BLOCKING: Added X-Content-Type-Options (nosniff), X-Frame-Options

   and '.ts' source files.

/**

import * as crypto from 'crypto';
import { prisma, pool } from '../db';
import { Role, AccountStatus } from '@prisma/client';

export const SESSION_COOKIE_NAME = 'ctvill_session';
export const CSRF_COOKIE_NAME = 'ctvill_csrf';

// Default session lifetimes
export const DEFAULT_SESSION_TTL_MS = 8 * 60 * 60 * 1000;    // 8 hours standard
export const REMEMBER_ME_SESSION_TTL_MS = 24 * 60 * 60 * 1000; // 24 hours Remember Me

export interface SessionData {
  id: string;
  userId: string;
  expiresAt: Date;
  createdAt: Date;
  lastActivityAt: Date;
  userAgent?: string | null;
  ipAddress?: string | null;
}

export interface VerifiedSessionUser {
  id: string;
  email: string;
  name: string;
  role: Role;
  accountStatus: AccountStatus;
  projectIds: string[];
}

/**
 * Hash opaque raw session token using SHA-256 before storing or looking up in PostgreSQL.
 */
export function hashSessionToken(rawToken: string): string {
  return crypto.createHash('sha256').update(rawToken).digest('hex');
}

/**
 * Generate a cryptographically secure random session token.
 */
export function generateSessionToken(): string {
  return crypto.randomBytes(32).toString('hex');
}

/**
 * Generate a cryptographically secure random CSRF token.
 */
export function generateCsrfToken(): string {
  return crypto.randomBytes(24).toString('hex');
}

/**
 * Create a new server-side session backed by PostgreSQL.
 */
export async function createSession(
  userId: string,
  options: {
    rememberMe?: boolean;
    userAgent?: string;
    ipAddress?: string;
  } = {}
): Promise<{ rawToken: string; expiresAt: Date; session: SessionData }> {
  const rawToken = generateSessionToken();
  const tokenHash = hashSessionToken(rawToken);
  const ttl = options.rememberMe ? REMEMBER_ME_SESSION_TTL_MS : DEFAULT_SESSION_TTL_MS;
  const expiresAt = new Date(Date.now() + ttl);

  const session = await prisma.session.create({
    data: {
      tokenHash,
      userId,
      expiresAt,
      userAgent: options.userAgent || null,
      ipAddress: options.ipAddress || null,
    },
  });

  return {
    rawToken,
    expiresAt,
    session: {
      id: session.id,
      userId: session.userId,
      expiresAt: session.expiresAt,
      createdAt: session.createdAt,
      lastActivityAt: session.lastActivityAt,
      userAgent: session.userAgent,
      ipAddress: session.ipAddress,
    },
  };
}

/**
 * Verify a session by its raw token and load the active user.
 * Returns null if token is missing, expired, not found, or user is not ACTIVE.
 */
export async function verifySession(rawToken: string | undefined): Promise<{
  user: VerifiedSessionUser;
  session: SessionData;
} | null> {
  if (!rawToken || typeof rawToken !== 'string' || rawToken.length < 16) {
    return null;
  }

  const tokenHash = hashSessionToken(rawToken.trim());
  const now = new Date();

  // Find session with associated user
  const record = await prisma.session.findUnique({
    where: { tokenHash },
    include: {
      user: {
        select: {
          id: true,
          email: true,
          name: true,
          role: true,
          accountStatus: true,
          projectIds: true,
        },
      },
    },
  });

  if (!record) {
    return null;
  }

  // Check expiration
  if (record.expiresAt <= now) {
    // Purge expired session asynchronously
    prisma.session.delete({ where: { id: record.id } }).catch(() => {});
    return null;
  }

  // Check user status: do not allow suspended or invited accounts
  if (!record.user || record.user.accountStatus !== AccountStatus.ACTIVE) {
    return null;
  }

  // Touch last activity pulse if it has been more than 60 seconds
  const lastActive = record.lastActivityAt.getTime();
  if (now.getTime() - lastActive > 60000) {
    prisma.session.update({
      where: { id: record.id },
      data: { lastActivityAt: now },
    }).catch(() => {});
  }

  return {
    user: {
      id: record.user.id,
      email: record.user.email,
      name: record.user.name,
      role: record.user.role,
      accountStatus: record.user.accountStatus,
      projectIds: record.user.projectIds || [],
    },
    session: {
      id: record.id,
      userId: record.userId,
      expiresAt: record.expiresAt,
      createdAt: record.createdAt,
      lastActivityAt: record.lastActivityAt,
      userAgent: record.userAgent,
      ipAddress: record.ipAddress,
    },
  };
}

/**
 * Invalidate a single active session (e.g. upon logout).
 */
export async function destroySession(rawToken: string | undefined): Promise<boolean> {
  if (!rawToken || typeof rawToken !== 'string') return false;
  try {
    const tokenHash = hashSessionToken(rawToken.trim());
    await prisma.session.deleteMany({
      where: { tokenHash },
    });
    return true;
  } catch {
    return false;
  }
}

/**
 * Revoke all sessions for a specific user (e.g. upon password reset, role change, or account suspension).
 */
export async function revokeAllUserSessions(userId: string): Promise<number> {
  try {
    const result = await prisma.session.deleteMany({
      where: { userId },
    });
    return result.count;
  } catch {
    return 0;
  }
}

/**
 * Periodic clean-up of expired sessions from PostgreSQL.
 */
export async function cleanupExpiredSessions(): Promise<number> {
  try {
    const res = await pool.query('DELETE FROM sessions WHERE expires_at <= CURRENT_TIMESTAMP');
    return res.rowCount || 0;
  } catch {
    return 0;
  }
}

import { Request, Response, NextFunction } from 'express';
import { SESSION_COOKIE_NAME, CSRF_COOKIE_NAME } from '../services/sessionService';

declare global {
  namespace Express {
    interface Request {
      cookies?: Record<string, string>;
    }
  }
}

/**
 * Robust zero-dependency cookie header parser
 */
export function parseCookieHeader(header: string | undefined): Record<string, string> {
  const cookies: Record<string, string> = {};
  if (!header || typeof header !== 'string') return cookies;

  const pairs = header.split(';');
  for (const pair of pairs) {
    const idx = pair.indexOf('=');
    if (idx < 0) continue;
    const rawKey = pair.substring(0, idx).trim();
    const rawVal = pair.substring(idx + 1).trim();
    if (!rawKey) continue;
    try {
      cookies[rawKey] = decodeURIComponent(rawVal);
    } catch {
      cookies[rawKey] = rawVal;
    }
  }
  return cookies;
}

/**
 * Express middleware to populate req.cookies
 */
export function cookieParserMiddleware(req: Request, _res: Response, next: NextFunction) {
  if (!req.cookies) {
    req.cookies = parseCookieHeader(req.headers.cookie);
  }
  next();
}

/**
 * Standard cookie configuration for authentication and CSRF
 */
export function getCookieOptions(options: {
  maxAgeMs?: number;
  httpOnly?: boolean;
  sameSite?: 'lax' | 'strict' | 'none';
} = {}) {
  const isProduction = process.env.NODE_ENV === 'production';
  return {
    httpOnly: options.httpOnly !== undefined ? options.httpOnly : true,
    secure: isProduction,
    sameSite: (options.sameSite || 'lax') as 'lax' | 'strict' | 'none',
    path: '/',
    maxAge: options.maxAgeMs !== undefined ? options.maxAgeMs : undefined,
  };
}

/**
 * Set the authentication session cookie
 */
export function setSessionCookie(res: Response, token: string, expiresAt: Date) {
  const maxAgeMs = Math.max(0, expiresAt.getTime() - Date.now());
  const options = getCookieOptions({
    maxAgeMs,
    httpOnly: true,
    sameSite: 'lax',
  });
  res.cookie(SESSION_COOKIE_NAME, token, options);
}

/**
 * Clear the authentication session cookie
 */
export function clearSessionCookie(res: Response) {
  const options = getCookieOptions({
    maxAgeMs: 0,
    httpOnly: true,
    sameSite: 'lax',
  });
  res.clearCookie(SESSION_COOKIE_NAME, options);
}

/**
 * Set the CSRF token cookie (readable by frontend client)
 */
export function setCsrfCookie(res: Response, token: string) {
  const options = getCookieOptions({
    httpOnly: false, // Must be readable by client to send back in X-CSRF-Token header
    sameSite: 'lax',
    maxAgeMs: 24 * 60 * 60 * 1000,
  });
  res.cookie(CSRF_COOKIE_NAME, token, options);
}

/**
 * Clear the CSRF token cookie
 */
export function clearCsrfCookie(res: Response) {
  const options = getCookieOptions({
    httpOnly: false,
    sameSite: 'lax',
    maxAgeMs: 0,
  });
  res.clearCookie(CSRF_COOKIE_NAME, options);
}

/**
 * @license
 * SPDX-License-Identifier: Apache-2.0
 */

import { Request, Response, NextFunction } from 'express';
import * as crypto from 'crypto';
import { CSRF_COOKIE_NAME, generateCsrfToken } from '../services/sessionService';
import { setCsrfCookie } from './cookies';

// Public endpoints that do not require CSRF token validation
const PUBLIC_CSRF_EXEMPT_PATHS = [
  '/api/auth/login',
  '/api/auth/verify-invite',
  '/api/auth/activate',
  '/api/quotations/submit',
  '/api/health',
];

/**
 * Ensures a CSRF cookie exists on every incoming request.
 */
export function ensureCsrfCookieMiddleware(req: Request, res: Response, next: NextFunction) {
  const currentToken = req.cookies?.[CSRF_COOKIE_NAME];
  if (!currentToken) {
    const newToken = generateCsrfToken();
    setCsrfCookie(res, newToken);
    if (!req.cookies) req.cookies = {};
    req.cookies[CSRF_COOKIE_NAME] = newToken;
  }
  next();
}

/**
 * Enforces CSRF token check on state-changing operations (POST, PUT, PATCH, DELETE)
 */
export function verifyCsrfMiddleware(req: Request, res: Response, next: NextFunction) {
  // Safe HTTP methods do not alter state
  if (['GET', 'HEAD', 'OPTIONS'].includes(req.method)) {
    return next();
  }

  // Exempt public onboarding and unauthenticated submission routes
  const normalizedPath = req.baseUrl ? `${req.baseUrl}${req.path}` : req.path;
  const isExempt = PUBLIC_CSRF_EXEMPT_PATHS.some(exemptPath => 
    normalizedPath === exemptPath || req.path === exemptPath || req.originalUrl?.split('?')[0] === exemptPath
  );

  if (isExempt) {
    return next();
  }

  // Header or body submitted token
  const submittedToken = (
    req.headers['x-csrf-token'] || 
    req.headers['csrf-token'] || 
    (req.body && typeof req.body === 'object' && req.body._csrf)
  ) as string | undefined;

  const expectedToken = req.cookies?.[CSRF_COOKIE_NAME];

  if (!submittedToken || !expectedToken) {
    return res.status(403).json({
      error: 'Forbidden: Missing CSRF security token. Please refresh the page and try again.',
      code: 'CSRF_MISSING'
    });
  }

  // Timing safe comparison
  const submittedBuf = Buffer.from(String(submittedToken).trim());
  const expectedBuf = Buffer.from(String(expectedToken).trim());

  if (submittedBuf.length !== expectedBuf.length || !crypto.timingSafeEqual(submittedBuf, expectedBuf)) {
    return res.status(403).json({
      error: 'Forbidden: Invalid CSRF security token mismatch.',
      code: 'CSRF_INVALID'
    });
  }

  next();
}

import { Request, Response, NextFunction } from 'express';

// List of allowed frontend origins for CORS with credentials
const DEFAULT_ALLOWED_ORIGINS = [
  'http://localhost:3000',
  'http://localhost:3001',
  'http://127.0.0.1:3000',
  'http://127.0.0.1:3001',
];

/**
 * Configure secure CORS with credentials support and explicit origin whitelisting.
 */
export function corsMiddleware(req: Request, res: Response, next: NextFunction) {
  const requestOrigin = req.headers.origin;
  const envOrigins = process.env.ALLOWED_ORIGINS 
    ? process.env.ALLOWED_ORIGINS.split(',').map(s => s.trim()) 
    : [];
  const allowedOrigins = [...DEFAULT_ALLOWED_ORIGINS, ...envOrigins];

  if (requestOrigin && allowedOrigins.includes(requestOrigin)) {
    res.header('Access-Control-Allow-Origin', requestOrigin);
    res.header('Access-Control-Allow-Credentials', 'true');
  } else if (!requestOrigin) {
    // Non-browser or same-origin direct requests
    res.header('Access-Control-Allow-Origin', '*');
  }

  res.header('Access-Control-Allow-Methods', 'GET, POST, PUT, DELETE, PATCH, OPTIONS');
  res.header(
    'Access-Control-Allow-Headers',
    'Origin, X-Requested-With, Content-Type, Accept, Authorization, x-csrf-token'
  );

  if (req.method === 'OPTIONS') {
    return res.sendStatus(204);
  }
  next();
}

/**
 * Standard OWASP security response headers
 */
export function securityHeadersMiddleware(_req: Request, res: Response, next: NextFunction) {
  res.setHeader('X-Content-Type-Options', 'nosniff');
  res.setHeader('X-Frame-Options', 'SAMEORIGIN');
  res.setHeader('X-XSS-Protection', '0');
  res.setHeader('Referrer-Policy', 'strict-origin-when-cross-origin');
  res.setHeader('Permissions-Policy', 'camera=(), microphone=(), geolocation=()');

  if (process.env.NODE_ENV === 'production') {
    res.setHeader('Strict-Transport-Security', 'max-age=31536000; includeSubDomains');
  }
  next();
}

/**
 * Prevent access to sensitive dotfiles, source files, and configuration files.
 */
export function preventSensitiveFileExposure(req: Request, res: Response, next: NextFunction) {
  const normalized = req.path.toLowerCase();
  if (
    normalized.includes('/.env') ||
    normalized.includes('/.git') ||
    normalized.endsWith('.env') ||
    normalized.endsWith('.prisma') ||
    normalized.endsWith('.ts') ||
    normalized.includes('/node_modules') ||
    normalized.includes('/prisma/')
  ) {
    return res.status(404).send('Not Found');
  }
  next();
}

/**
 * Lightweight, zero-dependency in-memory rate limiter
 */
interface RateLimitBucket {
  count: number;
  resetTime: number;
}

export function createRateLimiter(options: {
  windowMs: number;
  max: number;
  message?: string;
}) {
  const { windowMs, max, message } = options;
  const store = new Map<string, RateLimitBucket>();

  // Cleanup every 5 minutes
  setInterval(() => {
    const now = Date.now();
    for (const [key, bucket] of store.entries()) {
      if (bucket.resetTime <= now) {
        store.delete(key);
      }
    }
  }, 5 * 60 * 1000);

  return (req: Request, res: Response, next: NextFunction) => {
    // Skip rate limiting during test executions to prevent false 429 failures across assertion suites
    if (process.env.NODE_ENV === 'test') {
      return next();
    }

    // Key by IP address
    const ip = (req.headers['x-forwarded-for'] as string)?.split(',')[0].trim() || req.socket.remoteAddress || 'unknown';
    const now = Date.now();
    let bucket = store.get(ip);

    if (!bucket || bucket.resetTime <= now) {
      bucket = { count: 1, resetTime: now + windowMs };
      store.set(ip, bucket);
      return next();
    }

    bucket.count += 1;
    if (bucket.count > max) {
      const retryAfter = Math.ceil((bucket.resetTime - now) / 1000);
      res.setHeader('Retry-After', String(retryAfter));
      return res.status(429).json({
        error: message || 'Too many requests. Please try again later.',
        retryAfterSeconds: retryAfter,
      });
    }

    next();
  };
}

// 10 login attempts per 5 minutes
export const loginRateLimiter = createRateLimiter({

  max: 10,
  message: 'Too many authentication attempts. Please wait 5 minutes before trying again.',

  max: 20,
  message: 'Too many requests for this operation. Please wait a few minutes.',

export const CSRF_COOKIE_NAME = 'ctvill_csrf';

/**
 * Read the CSRF token from browser cookies
 */
export function getCsrfToken(): string | null {
  if (typeof document === 'undefined') return null;
  const match = document.cookie.match(new RegExp(`(^|;\\s*)(${CSRF_COOKIE_NAME})=([^;]*)`));
  return match ? decodeURIComponent(match[3]) : null;
}

// Global authorization failure callback
let authExpiredHandler: (() => void) | null = null;

export function setAuthExpiredHandler(handler: () => void) {
  authExpiredHandler = handler;
}

/**
 * Intercepts global window.fetch to ensure:
 * 1. Credentials (HttpOnly session cookies) are always included with /api requests
 * 2. CSRF tokens are automatically added to state-changing operations (POST, PUT, PATCH, DELETE)
 * 3. 401 Unauthorized responses trigger session cleanup and redirect to login
 */
let isFetchInitialized = false;

export function setupSecureFetch(onUnauthorized?: () => void) {
  if (typeof window === 'undefined' || isFetchInitialized) return;
  isFetchInitialized = true;

  if (onUnauthorized) {
    authExpiredHandler = onUnauthorized;
  }

  const originalFetch = window.fetch.bind(window);

  window.fetch = async function (input: RequestInfo | URL, init?: RequestInit): Promise<Response> {
    const rawUrl = typeof input === 'string' 
      ? input 
      : input instanceof Request 
      ? input.url 
      : input.toString();

    const isApiRequest = rawUrl.startsWith('/api') || rawUrl.includes('/api/');

    if (isApiRequest) {
      init = init ? { ...init } : {};
      
      // Always include session cookie credentials with API requests
      if (!init.credentials) {
        init.credentials = 'include';
      }

      // Check method
      const method = (
        init.method || 
        (input instanceof Request ? input.method : 'GET')
      ).toUpperCase();

      // Attach CSRF token on state-changing requests
      if (['POST', 'PUT', 'PATCH', 'DELETE'].includes(method)) {
        const csrfToken = getCsrfToken();
        if (csrfToken) {
          const headers = new Headers(init.headers || (input instanceof Request ? input.headers : undefined));
          if (!headers.has('x-csrf-token') && !headers.has('X-CSRF-Token')) {
            headers.set('x-csrf-token', csrfToken);
          }
          init.headers = headers;
        }
      }
    }

    try {
      const response = await originalFetch(input, init);

      // Handle session expiration on protected endpoints
      if (
        response.status === 401 && 
        isApiRequest && 
        !rawUrl.includes('/api/auth/login') &&
        !rawUrl.includes('/api/auth/verify-invite') &&
        !rawUrl.includes('/api/auth/activate')
      ) {
        if (authExpiredHandler) {
          authExpiredHandler();
        }
      }

      return response;
    } catch (error) {
      throw error;
    }
  };
}

/**
 * Standalone secure fetch helper
 */
export async function secureFetch(input: RequestInfo | URL, init?: RequestInit): Promise<Response> {
  return window.fetch(input, init);
}

// Create test application instance
function createTestApp() {
  const app = express();
  app.use(securityHeadersMiddleware);
  app.use(corsMiddleware);
  app.use(preventSensitiveFileExposure);
  app.use(cookieParserMiddleware);
  app.use(express.json());
  app.use(ensureCsrfCookieMiddleware);
  app.use(authenticateRequest);
  app.use('/api', verifyCsrfMiddleware);

  app.get('/api/health', (_req, res) => res.json({ status: 'ok' }));
  app.use('/api', authRouter);
  app.use('/api', projectsRouter);
  app.use('/api/payroll', payrollRouter);
  app.use('/api/attendance', attendanceRouter);

  return app;
}

interface TestResult {
  name: string;
  passed: boolean;
  error?: string;
}

const results: TestResult[] = [];

async function runTest(name: string, fn: () => Promise<void>) {
  process.stdout.write(`  Testing: ${name}... `);
  try {
    await fn();
    console.log('✅ PASSED');
    results.push({ name, passed: true });
  } catch (err: any) {
    console.log(`❌ FAILED: ${err.message}`);
    results.push({ name, passed: false, error: err.message });
  }
}

// Helper to make requests with cookies and headers
async function request(
  baseUrl: string,
  path: string,
  options: {
    method?: string;
    headers?: Record<string, string>;
    body?: any;
    cookie?: string;
    csrfToken?: string;
  } = {}
) {
  const url = `${baseUrl}${path}`;
  const headers: Record<string, string> = { ...options.headers };

  if (options.body && !headers['Content-Type']) {
    headers['Content-Type'] = 'application/json';
  }
  if (options.cookie) {
    headers['Cookie'] = options.cookie;
  }
  if (options.csrfToken) {
    headers['X-CSRF-Token'] = options.csrfToken;
  }

  const res = await fetch(url, {
    method: options.method || 'GET',
    headers,
    body: options.body ? JSON.stringify(options.body) : undefined,
  });

  const text = await res.text();
  let json: any = null;
  try {
    json = JSON.parse(text);
  } catch {}

  // Parse cookies from response
  const setCookie = res.headers.get('set-cookie');
  let sessionCookie: string | undefined;
  let csrfCookie: string | undefined;

  if (setCookie) {
    const sessionMatch = setCookie.match(new RegExp(`${SESSION_COOKIE_NAME}=([^;]+)`));
    if (sessionMatch) sessionCookie = sessionMatch[1];
    const csrfMatch = setCookie.match(new RegExp(`${CSRF_COOKIE_NAME}=([^;]+)`));
    if (csrfMatch) csrfCookie = csrfMatch[1];
  }

  return {
    status: res.status,
    headers: res.headers,
    json,
    text,
    sessionCookie,
    csrfCookie,
  };
}

async function main() {
  console.log('\n🔒 ==============================================================');
  console.log('   CTVILL ERP — PHASE 1 SECURITY & AUTHENTICATION TEST SUITE');
  console.log('==============================================================\n');

  await initializeDatabaseTables();

  const app = createTestApp();
  const server = http.createServer(app);

  await new Promise<void>((resolve) => {
    server.listen(0, '127.0.0.1', () => resolve());
  });

  const port = (server.address() as AddressInfo).port;
  const baseUrl = `http://127.0.0.1:${port}`;

  // Ensure test users exist in DB with known passwords
  const testAdminEmail = 'test.admin.phase1@ctvill.internal';
  const testTimekeeperEmail = 'test.timekeeper.phase1@ctvill.internal';
  const testFinanceEmail = 'test.finance.phase1@ctvill.internal';
  const testEngineerEmail = 'test.engineer.phase1@ctvill.internal';
  const testSuspendedEmail = 'test.suspended.phase1@ctvill.internal';
  const testPassword = 'Password123!';

  // Clean up old test accounts safely
  await prisma.user.deleteMany({
    where: { email: { in: [testAdminEmail, testTimekeeperEmail, testFinanceEmail, testEngineerEmail, testSuspendedEmail] } }
  }).catch(() => {});

  const testAdmin = await prisma.user.create({
    data: {
      email: testAdminEmail,
      name: 'Test Administrator',
      role: Role.ADMIN,
      accountStatus: AccountStatus.ACTIVE,
      passwordHash: hashPassword(testPassword),
      projectIds: [],
    }
  });

  const testTimekeeper = await prisma.user.create({
    data: {
      email: testTimekeeperEmail,
      name: 'Test Timekeeper',
      role: Role.TIMEKEEPER,
      accountStatus: AccountStatus.ACTIVE,
      passwordHash: hashPassword(testPassword),
      projectIds: ['PRJ-4693'],
    }
  });

  const testFinance = await prisma.user.create({
    data: {
      email: testFinanceEmail,
      name: 'Test Finance Officer',
      role: Role.FINANCE,
      accountStatus: AccountStatus.ACTIVE,
      passwordHash: hashPassword(testPassword),
      projectIds: [],
    }
  });

  const testEngineer = await prisma.user.create({
    data: {
      email: testEngineerEmail,
      name: 'Test Site Engineer',
      role: Role.ENGINEER,
      accountStatus: AccountStatus.ACTIVE,
      passwordHash: hashPassword(testPassword),
      projectIds: ['PRJ-4693'],
    }
  });

  const testSuspended = await prisma.user.create({
    data: {
      email: testSuspendedEmail,
      name: 'Test Suspended User',
      role: Role.ENGINEER,
      accountStatus: AccountStatus.SUSPENDED,
      passwordHash: hashPassword(testPassword),
      projectIds: [],
    }
  });

  try {
    // 1. Password Strength Validation
    await runTest('Password strength policy requires min 8 chars with letters & numbers', async () => {
      assert.strictEqual(isPasswordStrong('short').valid, false);
      assert.strictEqual(isPasswordStrong('alllettersnodigits').valid, false);
      assert.strictEqual(isPasswordStrong('12345678').valid, false);
      assert.strictEqual(isPasswordStrong('ValidPass123').valid, true);
    });

    // 2. Valid credentials authenticate and return session cookie
    let adminCookie = '';
    let adminCsrf = '';
    await runTest('Valid credentials authenticate active user and issue HttpOnly session cookie', async () => {
      const res = await request(baseUrl, '/api/auth/login', {
        method: 'POST',
        body: { email: testAdminEmail, password: testPassword },
      });
      assert.strictEqual(res.status, 200);
      assert.strictEqual(res.json?.success, true);
      assert.ok(res.json?.session);
      assert.strictEqual(res.json.session.role, 'Admin');
      assert.ok(res.sessionCookie, 'Session cookie must be set');
      assert.ok(res.csrfCookie || res.json.csrfToken, 'CSRF cookie or token must be issued');

      adminCookie = `${SESSION_COOKIE_NAME}=${res.sessionCookie}`;
      adminCsrf = res.csrfCookie || res.json.csrfToken;
    });

    // 3. Incorrect credentials rejected
    await runTest('Incorrect credentials rejected with generic error (no user enumeration)', async () => {
      const res = await request(baseUrl, '/api/auth/login', {
        method: 'POST',
        body: { email: testAdminEmail, password: 'WrongPassword999!' },
      });
      assert.strictEqual(res.status, 401);
      assert.strictEqual(res.json?.error, 'Invalid email or password.');

      // Non-existent email returns same generic error
      const nonExistent = await request(baseUrl, '/api/auth/login', {
        method: 'POST',
        body: { email: 'nonexistent999@ctvill.internal', password: 'AnyPassword123!' },
      });
      assert.strictEqual(nonExistent.status, 401);
      assert.strictEqual(nonExistent.json?.error, 'Invalid email or password.');
    });

    // 4. Requests without valid session return 401
    await runTest('Protected endpoints return 401 Unauthorized without session', async () => {
      const res = await request(baseUrl, '/api/auth/me');
      assert.strictEqual(res.status, 401);
      assert.strictEqual(res.json?.code, 'UNAUTHORIZED');

      const staffRes = await request(baseUrl, '/api/staff');
      assert.strictEqual(staffRes.status, 401);

      const projectsRes = await request(baseUrl, '/api/projects');
      assert.strictEqual(projectsRes.status, 401);
    });

    // 5. Forged x-user-role and x-user-id cannot elevate privileges
    await runTest('Forged x-user-role and x-user-id headers are ignored and cannot elevate privileges', async () => {
      const res = await request(baseUrl, '/api/staff', {
        headers: {
          'x-user-role': 'ADMIN',
          'x-user-id': testAdmin.id,
          'user-role': 'ADMIN',
        }
      });
      assert.strictEqual(res.status, 401, 'Request with forged headers must be rejected as 401');
    });

    // 6. User IDs in query or body cannot impersonate another user
    await runTest('Query parameter and body userId cannot impersonate another user', async () => {
      const res = await request(baseUrl, `/api/auth/profile?userId=${testAdmin.id}`);
      assert.strictEqual(res.status, 401, 'Unauthenticated query param userId must return 401');

      // Authenticate as timekeeper
      const tkRes = await request(baseUrl, '/api/auth/login', {
        method: 'POST',
        body: { email: testTimekeeperEmail, password: testPassword },
      });
      const tkCookie = `${SESSION_COOKIE_NAME}=${tkRes.sessionCookie}`;
      const tkCsrf = tkRes.csrfCookie || tkRes.json.csrfToken;

      // Timekeeper attempts to read profile passing admin's ID
      const spoofProfile = await request(baseUrl, `/api/auth/profile?userId=${testAdmin.id}`, {
        cookie: tkCookie,
      });
      assert.strictEqual(spoofProfile.status, 200);
      // Profile MUST be the authenticated timekeeper's profile, NOT the admin's!
      assert.strictEqual(spoofProfile.json?.profile?.id, testTimekeeper.id);
      assert.strictEqual(spoofProfile.json?.profile?.role, 'Timekeeper');
    });

    // 7. Unknown roles cannot become administrators
    await runTest('Unknown and unmapped roles reject access and NEVER normalize to ADMIN', async () => {
      assert.strictEqual(normalizeAppRole('HACKER'), null);
      assert.strictEqual(normalizeAppRole(''), null);
      assert.strictEqual(normalizeAppRole(null), null);
      assert.strictEqual(normalizeRole('INVALID_ROLE'), UserRole.UNAUTHORIZED);
      assert.strictEqual(normalizeRole(undefined), UserRole.UNAUTHORIZED);
    });

    // 8. Restricted operations return 403 for authenticated users without permission
    await runTest('Restricted operations return 403 Forbidden for unauthorized roles', async () => {
      // Timekeeper session
      const tkLogin = await request(baseUrl, '/api/auth/login', {
        method: 'POST',
        body: { email: testTimekeeperEmail, password: testPassword },
      });
      const tkCookie = `${SESSION_COOKIE_NAME}=${tkLogin.sessionCookie}; ${CSRF_COOKIE_NAME}=${tkLogin.csrfCookie}`;
      const tkCsrf = tkLogin.csrfCookie || tkLogin.json.csrfToken;

      // Timekeeper attempts to view staff list (ADMIN only)
      const staffRes = await request(baseUrl, '/api/staff', { cookie: tkCookie });
      assert.strictEqual(staffRes.status, 403);
    });

    // 9. Timekeeper cannot approve payroll or access unauthorized financial actions
    await runTest('Timekeeper cannot finalize/approve payroll (Separation of Duties)', async () => {
      const tkLogin = await request(baseUrl, '/api/auth/login', {
        method: 'POST',
        body: { email: testTimekeeperEmail, password: testPassword },
      });
      const tkCookie = `${SESSION_COOKIE_NAME}=${tkLogin.sessionCookie}; ${CSRF_COOKIE_NAME}=${tkLogin.csrfCookie}`;
      const tkCsrf = tkLogin.csrfCookie || tkLogin.json.csrfToken;

      const payrollRes = await request(baseUrl, '/api/payroll/finalize', {
        method: 'POST',
        cookie: tkCookie,
        csrfToken: tkCsrf,
        body: {
          projectId: 'PRJ-4693',
          periodStart: '2026-09-14',
          periodEnd: '2026-09-21',
        }
      });
      assert.strictEqual(payrollRes.status, 403);
      assert.strictEqual(payrollRes.json?.code, 'FORBIDDEN_SOD');
    });

    // 10. Finance users cannot create Administrator accounts
    await runTest('Finance users cannot create Administrator accounts (Admin only)', async () => {
      const finLogin = await request(baseUrl, '/api/auth/login', {
        method: 'POST',
        body: { email: testFinanceEmail, password: testPassword },
      });
      const finCookie = `${SESSION_COOKIE_NAME}=${finLogin.sessionCookie}; ${CSRF_COOKIE_NAME}=${finLogin.csrfCookie}`;
      const finCsrf = finLogin.csrfCookie || finLogin.json.csrfToken;

      const createAdminRes = await request(baseUrl, '/api/staff', {
        method: 'POST',
        cookie: finCookie,
        csrfToken: finCsrf,
        body: {
          email: 'rogue.admin@ctvill.internal',
          name: 'Rogue Admin',
          password: 'Password123!',
          role: 'Admin',
        }
      });
      assert.strictEqual(createAdminRes.status, 403);
    });

    // 11. Project assignment scoping checks
    await runTest('Users restricted by projectIds cannot access unauthorized projects', async () => {
      const user = {
        id: 'u1',
        email: 'eng@ctvill.com',
        name: 'Engineer',
        role: 'ENGINEER' as const,
        projectIds: ['PRJ-4693'],
      };

      assert.strictEqual(checkUserProjectAccess(user, 'PRJ-4693'), true);
      assert.strictEqual(checkUserProjectAccess(user, 'PRJ-OTHER-999'), false);

      // Admin has global access
      const admin = {
        id: 'u2',
        email: 'admin@ctvill.com',
        name: 'Admin',
        role: 'ADMIN' as const,
        projectIds: [],
      };
      assert.strictEqual(checkUserProjectAccess(admin, 'PRJ-OTHER-999'), true);
    });

    // 12. Suspended accounts are rejected
    await runTest('Suspended accounts cannot log in or access resources', async () => {
      const res = await request(baseUrl, '/api/auth/login', {
        method: 'POST',
        body: { email: testSuspendedEmail, password: testPassword },
      });
      assert.strictEqual(res.status, 403);
      assert.ok(res.json?.error?.toLowerCase().includes('suspended'));
    });

    // 13. Expired sessions are rejected
    await runTest('Expired sessions are rejected with 401', async () => {
      // Create session expired in the past
      const { rawToken } = await createSession(testAdmin.id, { rememberMe: false });
      const tokenHash = hashSessionToken(rawToken);
      await prisma.session.update({
        where: { tokenHash },
        data: { expiresAt: new Date(Date.now() - 10000) }
      });

      const res = await request(baseUrl, '/api/auth/me', {
        cookie: `${SESSION_COOKIE_NAME}=${rawToken}`,
      });
      assert.strictEqual(res.status, 401);
    });

    // 14. Logout invalidates the active server-side session
    await runTest('Logout invalidates active session in PostgreSQL and clears cookie', async () => {
      const loginRes = await request(baseUrl, '/api/auth/login', {
        method: 'POST',
        body: { email: testAdminEmail, password: testPassword },
      });
      const cookie = `${SESSION_COOKIE_NAME}=${loginRes.sessionCookie}; ${CSRF_COOKIE_NAME}=${loginRes.csrfCookie}`;
      const csrf = loginRes.csrfCookie || loginRes.json.csrfToken;

      // Verify authenticated
      const meRes = await request(baseUrl, '/api/auth/me', { cookie });
      assert.strictEqual(meRes.status, 200);

      // Logout
      const logoutRes = await request(baseUrl, '/api/auth/logout', {
        method: 'POST',
        cookie,
        csrfToken: csrf,
      });
      assert.strictEqual(logoutRes.status, 200);

      // Verify session is dead
      const postLogoutMe = await request(baseUrl, '/api/auth/me', { cookie });
      assert.strictEqual(postLogoutMe.status, 401);
    });

    // 15. Password changes invalidate existing sessions
    await runTest('Password change invalidates existing sessions', async () => {
      // Create session 1
      const login1 = await request(baseUrl, '/api/auth/login', {
        method: 'POST',
        body: { email: testEngineerEmail, password: testPassword },
      });
      const cookie1 = `${SESSION_COOKIE_NAME}=${login1.sessionCookie}; ${CSRF_COOKIE_NAME}=${login1.csrfCookie}`;
      const csrf1 = login1.csrfCookie || login1.json.csrfToken;

      // Change password
      const newPass = 'NewEngineerPass123!';
      const changeRes = await request(baseUrl, '/api/auth/change-password', {
        method: 'POST',
        cookie: cookie1,
        csrfToken: csrf1,
        body: {
          currentPassword: testPassword,
          newPassword: newPass,
        }
      });
      assert.strictEqual(changeRes.status, 200);

      // Old session 1 should now be revoked (or invalidated)
      const oldCheck = await request(baseUrl, '/api/auth/me', { cookie: cookie1 });
      assert.strictEqual(oldCheck.status, 401);

      // Can log in with new password
      const newLogin = await request(baseUrl, '/api/auth/login', {
        method: 'POST',
        body: { email: testEngineerEmail, password: newPass },
      });
      assert.strictEqual(newLogin.status, 200);
    });

    // 16. CSRF protection blocks unauthorized state-changing requests
    await runTest('CSRF protection blocks state-changing requests with missing or mismatched token', async () => {
      // Missing token
      const resNoCsrf = await request(baseUrl, '/api/projects', {
        method: 'POST',
        cookie: adminCookie,
        body: { name: 'Unauthorized CSRF Project', clientName: 'Attacker' }
      });
      assert.strictEqual(resNoCsrf.status, 403);
      assert.strictEqual(resNoCsrf.json?.code, 'CSRF_MISSING');

      // Mismatched token
      const resBadCsrf = await request(baseUrl, '/api/projects', {
        method: 'POST',
        cookie: adminCookie,
        csrfToken: 'forged_fake_csrf_token_1234567890',
        body: { name: 'Unauthorized CSRF Project', clientName: 'Attacker' }
      });
      assert.strictEqual(resBadCsrf.status, 403);
    });

    // 17. Existing database records and demo users are preserved
    await runTest('Existing database users and data are intact and preserved', async () => {
      const realAdmin = await prisma.user.findFirst({ where: { email: 'admin@ctvill.com' } });
      assert.ok(realAdmin, 'admin@ctvill.com must exist in database');

      const realOM = await prisma.user.findFirst({ where: { email: 'om@ctvill.com' } });
      assert.ok(realOM, 'om@ctvill.com must exist in database');

      const realFinance = await prisma.user.findFirst({ where: { email: 'finance@ctvill.com' } });
      assert.ok(realFinance, 'finance@ctvill.com must exist in database');

      const realTimekeeper = await prisma.user.findFirst({ where: { email: 'timekeeper@ctvill.com' } });
      assert.ok(realTimekeeper, 'timekeeper@ctvill.com must exist in database');

      const projectCount = await prisma.commercialProject.count();
      assert.ok(projectCount >= 1, 'Commercial projects must be preserved');
    });

  } finally {
    // Clean up test accounts safely
    await prisma.user.deleteMany({
      where: { email: { in: [testAdminEmail, testTimekeeperEmail, testFinanceEmail, testEngineerEmail, testSuspendedEmail] } }
    }).catch(() => {});

    server.close();
  }

  console.log('\n==============================================================');
  const passed = results.filter(r => r.passed).length;
  const failed = results.filter(r => !r.passed).length;
  console.log(`SUMMARY: ${passed} passed, ${failed} failed out of ${results.length} total tests.`);
  console.log('==============================================================\n');

  if (failed > 0) {
    process.exit(1);
  }
}

main().catch((err) => {
  console.error('Test execution failed:', err);
  process.exit(1);
});
